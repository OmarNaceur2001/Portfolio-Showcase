def generate_demo_data(n: int = 1000, seed: int = 42) -> "pd.DataFrame":
    """Synthetic classification dataset with 5 features + label."""
    rng = np.random.default_rng(seed)
    X = rng.standard_normal((n, 5))
    noise = rng.standard_normal(n) * 0.3
    score = 0.4*X[:,0] - 0.6*X[:,1] + 0.2*X[:,2] + noise
    y = (score > score.mean()).astype(int)

    df = pd.DataFrame(X, columns=[f"feature_{i}" for i in range(1, 6)])
    df["label"] = y
    df["group"] = rng.choice(["A", "B", "C"], size=n, p=[0.5, 0.3, 0.2])
    return df

def profile_dataset(df: "pd.DataFrame", name: str = "dataset") -> dict:
    report = {
        "dataset_name": name,
        "generated_at": str(datetime.datetime.utcnow()),
        "rows": len(df),
        "columns": len(df.columns),
        "memory_kb": round(df.memory_usage(deep=True).sum() / 1024, 2),
        "missing_cells": int(df.isnull().sum().sum()),
        "duplicate_rows": int(df.duplicated().sum()),
        "column_details": {},
    }

    for col in df.columns:
        s = df[col]
        dtype = str(s.dtype)
        missing = int(s.isnull().sum())
        detail = {
            "dtype": dtype,
            "missing": missing,
            "missing_pct": round(missing / len(df) * 100, 2),
        }
        if pd.api.types.is_numeric_dtype(s):
            detail.update({
                "min": round(float(s.min()), 4),
                "max": round(float(s.max()), 4),
                "mean": round(float(s.mean()), 4),
                "std": round(float(s.std()), 4),
                "median": round(float(s.median()), 4),
                "skew": round(float(s.skew()), 4),
            })
        else:
            vc = s.value_counts()
            detail.update({
                "unique": int(s.nunique()),
                "top_values": vc.head(5).to_dict(),
            })
        report["column_details"][col] = detail

    return report


def print_profile_report(report: dict):
    line = "─" * 60
    print(f"\n{'═'*60}")
    print(f"  DATA PROFILE  →  {report['dataset_name']}")
    print(f"{'═'*60}")
    print(f"  Rows          : {report['rows']:,}")
    print(f"  Columns       : {report['columns']}")
    print(f"  Memory        : {report['memory_kb']} KB")
    print(f"  Missing cells : {report['missing_cells']}")
    print(f"  Duplicate rows: {report['duplicate_rows']}")
    print(line)
    for col, d in report["column_details"].items():
        print(f"\n  [{col}]  dtype={d['dtype']}  missing={d['missing_pct']}%")
        if "mean" in d:
            print(f"    min={d['min']}  max={d['max']}  mean={d['mean']}  "
                  f"std={d['std']}  skew={d['skew']}")
        else:
            print(f"    unique={d['unique']}  top={list(d['top_values'].items())[:3]}")
    print(f"\n{'═'*60}\n")


# ═══════════════════════════════════════════════════════════
#  SECTION 4 – REPARTITIONING ENGINE
# ═══════════════════════════════════════════════════════════

class Repartitioner:
    """
    Supports three strategies:
      • random      – simple shuffle split
      • stratified  – preserves class distribution (requires label col)
      • kfold       – k-fold cross-validation splits
    """

    def __init__(self, df: "pd.DataFrame", profile: dict):
        self.df = df.copy().reset_index(drop=True)
        self.splits_cfg = profile["default_splits"]
        self.seed = profile["default_random_seed"]
        self.strategy = profile["preferred_strategy"]

    # ── helpers ────────────────────────────────────────────

    def _validate_ratios(self, train: float, val: float, test: float):
        total = round(train + val + test, 6)
        if not math.isclose(total, 1.0, rel_tol=1e-5):
            raise ValueError(f"Splits must sum to 1.0 (got {total})")

    def _checksum(self, part: "pd.DataFrame") -> str:
        h = hashlib.md5(pd.util.hash_pandas_object(part, index=True)
                        .values.tobytes()).hexdigest()
        return h[:8]

    # ── strategy: random ──────────────────────────────────

    def random_split(self,
                     train_r: float = None,
                     val_r: float = None,
                     test_r: float = None) -> dict:
        tr = train_r or self.splits_cfg["train"]
        vr = val_r  or self.splits_cfg["val"]
        te = test_r or self.splits_cfg["test"]
        self._validate_ratios(tr, vr, te)

        idx = np.random.default_rng(self.seed).permutation(len(self.df))
        n_tr = int(tr * len(self.df))
        n_vl = int(vr * len(self.df))

        train = self.df.iloc[idx[:n_tr]]
        val   = self.df.iloc[idx[n_tr:n_tr+n_vl]]
        test  = self.df.iloc[idx[n_tr+n_vl:]]
        return {"train": train, "val": val, "test": test}

    # ── strategy: stratified ──────────────────────────────

    def stratified_split(self,
                         label_col: str = "label",
                         train_r: float = None,
                         val_r: float = None,
                         test_r: float = None) -> dict:
        if not HAS_SKLEARN:
            print("[warn] scikit-learn not found; falling back to random split.")
            return self.random_split(train_r, val_r, test_r)

        tr = train_r or self.splits_cfg["train"]
        vr = val_r  or self.splits_cfg["val"]
        te = test_r or self.splits_cfg["test"]
        self._validate_ratios(tr, vr, te)

        if label_col not in self.df.columns:
            raise KeyError(f"Label column '{label_col}' not found in dataset.")

        y = self.df[label_col]
        # first split: train vs (val+test)
        X_tr, X_tmp, y_tr, y_tmp = train_test_split(
            self.df, y, test_size=(vr+te), random_state=self.seed, stratify=y
        )
        # second split: val vs test
        val_ratio_of_tmp = vr / (vr + te)
        X_val, X_te, _, _ = train_test_split(
            X_tmp, y_tmp, test_size=(1-val_ratio_of_tmp),
            random_state=self.seed, stratify=y_tmp
        )
        return {"train": X_tr, "val": X_val, "test": X_te}

    # ── strategy: k-fold ──────────────────────────────────

    def kfold_split(self,
                    k: int = 5,
                    label_col: str = "label",
                    stratified: bool = True) -> list:
        if not HAS_SKLEARN:
            print("[warn] scikit-learn not found; kfold unavailable.")
            return []

        y = self.df[label_col].values if label_col in self.df.columns else None
        splitter = (StratifiedKFold(n_splits=k, shuffle=True,
                                    random_state=self.seed)
                    if stratified and y is not None
                    else KFold(n_splits=k, shuffle=True,
                               random_state=self.seed))

        folds = []
        for fold_i, (tr_idx, te_idx) in enumerate(
                splitter.split(self.df, y), start=1):
            folds.append({
                "fold": fold_i,
                "train": self.df.iloc[tr_idx],
                "test": self.df.iloc[te_idx],
            })
        return folds

    # ── save partitions ───────────────────────────────────

    def save_splits(self, splits: dict, out_dir: str = "partitions"):
        os.makedirs(out_dir, exist_ok=True)
        for name, part in splits.items():
            path = os.path.join(out_dir, f"{name}.csv")
            part.to_csv(path, index=False)
        print(f"[save] Partitions written to  →  ./{out_dir}/")

    def save_folds(self, folds: list, out_dir: str = "folds"):
        os.makedirs(out_dir, exist_ok=True)
        for f in folds:
            k = f["fold"]
            f["train"].to_csv(os.path.join(out_dir, f"fold{k}_train.csv"),
                              index=False)
            f["test"].to_csv(os.path.join(out_dir, f"fold{k}_test.csv"),
                             index=False)
        print(f"[save] Folds written to  →  ./{out_dir}/")

    # ── print partition summary ───────────────────────────

    def print_split_summary(self, splits: dict, label_col: str = "label"):
        print(f"\n{'═'*60}")
        print(f"  PARTITION SUMMARY  (strategy: {self.strategy})")
        print(f"{'═'*60}")
        total = sum(len(v) for v in splits.values())
        for name, part in splits.items():
            pct = round(len(part)/total*100, 1)
            dist = ""
            if label_col in part.columns:
                vc = part[label_col].value_counts().to_dict()
                dist = f"  labels={vc}"
            chk = self._checksum(part)
            print(f"  {name:<8}  {len(part):>6} rows  ({pct}%)"
                  f"  md5={chk}{dist}")
        print(f"{'═'*60}\n")

    def print_fold_summary(self, folds: list, label_col: str = "label"):
        print(f"\n{'═'*60}")
        print(f"  K-FOLD SUMMARY  ({len(folds)} folds)")
        print(f"{'═'*60}")
        for f in folds:
            tr_n, te_n = len(f["train"]), len(f["test"])
            print(f"  Fold {f['fold']}  train={tr_n}  test={te_n}")
        print(f"{'═'*60}\n")

def parse_args():
    p = argparse.ArgumentParser(
        description="Data Repartitioner & Profile Generator"
    )
    p.add_argument("--csv",      type=str,   default=None,
                   help="Path to input CSV (default: generated demo data)")
    p.add_argument("--label",    type=str,   default="label",
                   help="Name of the target/label column")
    p.add_argument("--strategy", type=str,
                   choices=["random", "stratified", "kfold"],
                   default=None,
                   help="Split strategy (overrides profile default)")
    p.add_argument("--train",    type=float, default=None,
                   help="Train ratio  e.g. 0.70")
    p.add_argument("--val",      type=float, default=None,
                   help="Val ratio    e.g. 0.15")
    p.add_argument("--test",     type=float, default=None,
                   help="Test ratio   e.g. 0.15")
    p.add_argument("--k",        type=int,   default=5,
                   help="Number of folds for kfold strategy")
    p.add_argument("--no-save",  action="store_true",
                   help="Skip writing output CSV files")
    p.add_argument("--profile-name", type=str, default=None,
                   help="Override profile display name")
    return p.parse_args()

def main():
    if not HAS_PANDAS:
        print("[error] pandas / numpy not installed.  Run:  pip install pandas numpy")
        sys.exit(1)

    args = parse_args()

    # ── 1. Profile ─────────────────────────────────────────
    profile = load_or_create_profile()
    if args.profile_name:
        profile = update_profile(profile, "name", args.profile_name)
    banner(profile)

    # ── 2. Load data ───────────────────────────────────────
    if args.csv:
        print(f"[data] Loading CSV  →  {args.csv}")
        df = pd.read_csv(args.csv)
        dataset_name = os.path.basename(args.csv).replace(".csv", "")
    else:
        print("[data] No CSV provided — generating 1 000-row demo dataset …")
        df = generate_demo_data(n=1000, seed=profile["default_random_seed"])
        dataset_name = "demo_dataset"

    # ── 3. Profile the dataset ─────────────────────────────
    report = profile_dataset(df, name=dataset_name)
    print_profile_report(report)

    # save profile JSON
    report_path = f"{dataset_name}_profile.json"
    with open(report_path, "w") as f:
        json.dump(report, f, indent=2)
    print(f"[profile] Data profile saved  →  {report_path}")

    # ── 4. Repartition ─────────────────────────────────────
    strategy = args.strategy or profile["preferred_strategy"]
    rp = Repartitioner(df, profile)
    rp.strategy = strategy

    label_col = args.label

    if strategy == "kfold":
        print(f"[split] Running {args.k}-fold cross-validation …")
        folds = rp.kfold_split(k=args.k, label_col=label_col)
        if folds:
            rp.print_fold_summary(folds, label_col)
            if not args.no_save:
                rp.save_folds(folds)

    elif strategy == "stratified":
        print(f"[split] Running stratified split on column '{label_col}' …")
        try:
            splits = rp.stratified_split(
                label_col=label_col,
                train_r=args.train, val_r=args.val, test_r=args.test
            )
        except KeyError:
            print(f"[warn] Label column '{label_col}' missing; falling back to random.")
            splits = rp.random_split(args.train, args.val, args.test)
        rp.print_split_summary(splits, label_col)
        if not args.no_save:
            rp.save_splits(splits)

    else:  # random
        print("[split] Running random split …")
        splits = rp.random_split(args.train, args.val, args.test)
        rp.print_split_summary(splits, label_col)
        if not args.no_save:
            rp.save_splits(splits)

    # ── 5. Done ────────────────────────────────────────────
    print("[✓] Done.  Files in this directory:")
    for f in sorted(os.listdir(".")):
        if f.endswith((".json", ".csv")):
            size = os.path.getsize(f)
            print(f"    {f:<40}  {size:>8} bytes")

    # update last_run in profile
    update_profile(profile, "last_run", str(datetime.datetime.utcnow()))
    update_profile(profile, "last_dataset", dataset_name)


if __name__ == "__main__":
    main()
